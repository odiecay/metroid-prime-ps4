cmake_minimum_required(VERSION 3.25)
project(project_prime VERSION 0.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT PS4 OR NOT ORBIS_COMPAT_DIR)
  message(FATAL_ERROR "Use platform/ps4/toolchain.cmake for native Orbis builds")
endif()
orbis_compat_verify()
add_executable(prime_runtime_probe tests/runtime_probe.cpp)
set_target_properties(prime_runtime_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
target_link_libraries(prime_runtime_probe PRIVATE SceSystemService)
ps4_create_eboot(prime_runtime_probe)
option(PRIME_BUILD_BOOT_PROBE "Build a minimal native C startup control image" OFF)
if(PRIME_BUILD_BOOT_PROBE)
  add_executable(prime_boot_probe tests/boot_probe.c platform/ps4/startup.c)
  target_link_libraries(prime_boot_probe PRIVATE SceSystemService)
  set_target_properties(prime_boot_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/boot")
  ps4_create_eboot(prime_boot_probe)
endif()
add_subdirectory("${ORBIS_COMPAT_DIR}/vkloader" vkloader)
add_executable(prime_vulkan_probe tests/vulkan_probe.c)
set_target_properties(prime_vulkan_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/vulkan")
target_link_libraries(prime_vulkan_probe PRIVATE ps4-vkloader SceSystemService)
ps4_create_eboot(prime_vulkan_probe)
add_library(prime_gc_iso STATIC platform/ps4/disc/gc_iso.cpp)
target_include_directories(prime_gc_iso PUBLIC external/nod-host-package/include)
option(PRIME_BUILD_MODERN_PROBE "Link a probe using the modern native C++ facilities" OFF)
if(PRIME_BUILD_MODERN_PROBE)
  add_executable(prime_runtime_modern_probe tests/runtime_modern_probe.cpp)
  set_target_properties(prime_runtime_modern_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modern-runtime")
  ps4_create_eboot(prime_runtime_modern_probe)
endif()
option(PRIME_BUILD_GAME_OBJECTS "Compile the full upstream game manifest for Orbis" ON)
if(PRIME_BUILD_GAME_OBJECTS)
  include(platform/ps4/game-objects.cmake)
endif()
option(PRIME_BUILD_DAWN_PROBE "Build native WGSL/readback and Orbis presentation diagnostics" OFF)
if(PRIME_BUILD_DAWN_PROBE)
  add_library(prime_dawn STATIC IMPORTED)
  set_target_properties(prime_dawn PROPERTIES
    IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/build/dawn-modern/src/dawn/native/libwebgpu_dawn.a"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/external/dawn/include;${CMAKE_SOURCE_DIR}/build/dawn-modern/gen/include")
  add_executable(prime_dawn_probe tests/dawn_probe.cpp)
  set_target_properties(prime_dawn_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dawn")
  target_link_libraries(prime_dawn_probe PRIVATE prime_dawn ps4-vkloader SceSystemService)
  ps4_create_eboot(prime_dawn_probe)
endif()
option(PRIME_BUILD_SIGNAL_PROBE "Build an isolated native signal ABI/delivery test" OFF)
if(PRIME_BUILD_SIGNAL_PROBE)
  add_executable(prime_signal_probe tests/signal_probe.cpp platform/ps4/crash.cpp)
  target_include_directories(prime_signal_probe PRIVATE game/MetroidPrimePort/platform/include)
  set_target_properties(prime_signal_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/signal")
  ps4_create_eboot(prime_signal_probe)
endif()
option(PRIME_BUILD_STORAGE_PROBE "Build native file-lock/SQLite persistence checks" OFF)
if(PRIME_BUILD_STORAGE_PROBE)
  add_library(prime_sqlite STATIC IMPORTED)
  set_target_properties(prime_sqlite PROPERTIES
    IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/build/game-native/upstream/extern/aurora/extern/libsqlite3.a"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/build/game-native/_deps/sqlite3-src")
  add_executable(prime_storage_probe tests/storage_probe.c)
  target_compile_options(prime_storage_probe PRIVATE
    -include "${CMAKE_SOURCE_DIR}/platform/ps4/compat/sqlite_orbis_fcntl.h")
  target_link_libraries(prime_storage_probe PRIVATE prime_sqlite)
  set_target_properties(prime_storage_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/storage")
  ps4_create_eboot(prime_storage_probe)
endif()
