cmake_minimum_required(VERSION 3.25)
project(prime_sdl LANGUAGES C CXX)
if(NOT PS4)
  message(FATAL_ERROR "Native Orbis toolchain required")
endif()
set(SDL_PRIME_PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Project Prime Orbis SDL implementation" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
set(SDL_AVX OFF CACHE BOOL "" FORCE)
set(SDL_AVX2 OFF CACHE BOOL "" FORCE)
set(SDL_AVX512F OFF CACHE BOOL "" FORCE)
unset(SDL_JOYSTICK_VIRTUAL CACHE)
set(SDL_VIRTUAL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_PTHREADS ON CACHE BOOL "" FORCE)
set(SDL_DUMMYAUDIO OFF CACHE BOOL "" FORCE)
set(SDL_DYNAPI OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../external/SDL-release-3.4.10" sdl)
target_compile_options(SDL3-static PRIVATE -fshort-wchar)
add_library(prime_pad STATIC orbis_pad.c)
target_link_libraries(prime_pad PUBLIC SDL3::SDL3-static ScePad SceUserService)
target_include_directories(prime_pad PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
