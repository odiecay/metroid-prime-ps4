cmake_minimum_required(VERSION 3.25)
project(prime_pc_asset_importer LANGUAGES C CXX)
# PC-only preparation utility. No object from this build is linked into the PKG.
if(PS4 OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "The asset preparation tool is a host build")
endif()
get_filename_component(PRIME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(DAWN_FETCH_DEPENDENCIES ON CACHE BOOL "" FORCE)
set(DAWN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DAWN_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(DAWN_BUILD_PROTOBUF OFF CACHE BOOL "" FORCE)
set(DAWN_USE_GLFW OFF CACHE BOOL "" FORCE)
set(DAWN_USE_X11 OFF CACHE BOOL "" FORCE)
set(DAWN_USE_WAYLAND OFF CACHE BOOL "" FORCE)
# The command-line import exits before GPU/window initialization. Build only
# Dawn's CPU helper backend here; native PS4 keeps its separate real Vulkan build.
set(DAWN_ENABLE_NULL ON CACHE BOOL "" FORCE)
set(DAWN_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_DESKTOP_GL OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_OPENGLES OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_SPIRV_VALIDATION OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_SPV_READER OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_SPV_WRITER OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_GLSL_WRITER OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_GLSL_VALIDATOR OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_NULL_WRITER ON CACHE BOOL "" FORCE)
set(TINT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TINT_BUILD_CMD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PRIME_ROOT}/external/dawn" dawn)
if(NOT TARGET dawn::webgpu_dawn)
  add_library(dawn::webgpu_dawn ALIAS webgpu_dawn)
endif()
set(Dawn_FOUND TRUE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
foreach(prime_sdl_feature X11 WAYLAND KMSDRM ALSA PULSEAUDIO JACK PIPEWIRE OPENGL OPENGLES VULKAN)
  set(SDL_${prime_sdl_feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${PRIME_ROOT}/external/SDL-release-3.4.10" sdl)
set(SDL3_FOUND TRUE)
list(PREPEND CMAKE_PREFIX_PATH "${PRIME_ROOT}/external/nod-host-package")
set(AURORA_DAWN_PROVIDER system CACHE STRING "" FORCE)
set(AURORA_DAWN_LINKAGE static CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER system CACHE STRING "" FORCE)
set(AURORA_SDL3_LINKAGE static CACHE STRING "" FORCE)
set(AURORA_NOD_PROVIDER system CACHE STRING "" FORCE)
set(MP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MP_EMBED_RESOURCES OFF CACHE BOOL "" FORCE)
set(MP_USE_MOLD OFF CACHE BOOL "" FORCE)
set(MP_USE_CCACHE OFF CACHE BOOL "" FORCE)
set(MP_BUNDLED_DEPS ON CACHE BOOL "" FORCE)
add_subdirectory("${PRIME_ROOT}/game/MetroidPrimePort" importer)

# Original menu preparation only; reuse the upstream JPEG/THP writer and the
# same bounded portable decoder used by the native game. No runtime changes.
add_executable(prime_menu_movie
  "${PRIME_ROOT}/tools/prepare_menu_movie.cpp"
  "${PRIME_ROOT}/game/MetroidPrimePort/platform/port_remastered_movie.cpp"
  "${PRIME_ROOT}/game/MetroidPrimePort/extern/aurora/lib/dolphin/thp/THPDec.cpp")
target_include_directories(prime_menu_movie PRIVATE
  "${PRIME_ROOT}/game/MetroidPrimePort/platform/include"
  "${PRIME_ROOT}/game/MetroidPrimePort/extern/aurora/include")
target_compile_definitions(prime_menu_movie PRIVATE TARGET_PC)
target_link_libraries(prime_menu_movie PRIVATE fmt::fmt)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_compile_options(prime_menu_movie PRIVATE -O2 -msse4.2 -mpopcnt -mcx16 -mtune=btver2 -mno-avx)
endif()
